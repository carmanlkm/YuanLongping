Google Apps Script (GAS) 複製碼： 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["時間", "班別", "學號", "勝出局數", "首答答對數", "Q1首答", "Q2首答", "Q3首答", "Q4首答", "Q5首答", "Q6首答", "Q7首答", "Q8首答", "Q9首答", "Q10首答", "Q11首答", "Q12首答"]);
  }
  var row = [new Date(), data.studentClass, data.studentID, data.winsCount, data.firstCorrectCount];
  for (var i = 1; i <= 12; i++) {
    var q = data.firstAttempts ? data.firstAttempts[i] : null;
    row.push(q ? (q.isCorrect ? "正確" : "錯誤(選項"+q.selectedOption+")") : "未答");
  }
  sheet.appendRow(row);
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}

教學：在 Google Sheet 選「擴充功能」->「Apps Script」，貼上代碼，部署為「Web 應用程式」(存取權限設為「所有人」) 並複製 URL 即可！
